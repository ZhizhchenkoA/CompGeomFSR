41 const double tolerance = std::abs(eps);
42 const Point r = subtract(b, a);
43 const Point s = subtract(d, c);
44 const double rLength = length(r);
45 const double sLength = length(s);
46
47 // Вырожденные отрезки: точка-точка и точка-отрезок.
48 if (rLength <= tolerance && sLength <= tolerance) {
49 return onSegment(a, a, c, tolerance)
50 ? SegmentIntersection{SegmentIntersectionKind::Point, a, {}}
51 : SegmentIntersection{};
52 }
53 if (rLength <= tolerance) {
54 return onSegment(c, d, a, tolerance)
55 ? SegmentIntersection{SegmentIntersectionKind::Point, a, {}}
56 : SegmentIntersection{};
57 }
58 if (sLength <= tolerance) {
59 return onSegment(a, b, c, tolerance)
60 ? SegmentIntersection{SegmentIntersectionKind::Point, c, {}}
61 : SegmentIntersection{};
62 }
63
64 const Point cMinusA = subtract(c, a);
65 const double rCrossS = crossVectors(r, s);
66 // cross(r, s) имеет размерность площади; допускаем отклонение,
67 // соответствующее полосе шириной tolerance около более длинного отрезка.
68 const double parallelTolerance = tolerance * std::max(rLength, sLength);
69
70 if (std::abs(rCrossS) <= parallelTolerance) {
71 // Параллельные отрезки пересекаются только при коллинеарности.
72 if (std::abs(crossVectors(cMinusA, r)) > tolerance * rLength) {
73 return {};
74 }
75
76 const double rLengthSquared = dot(r, r);
77 const double tC = dot(cMinusA, r) / rLengthSquared;
78 const double tD = dot(subtract(d, a), r) / rLengthSquared;
79 const double from = std::max(0.0, std::min(tC, tD));
80 const double to = std::min(1.0, std::max(tC, tD));
81 const double parameterTolerance = tolerance / rLength;
82 if (from > to + parameterTolerance) {
83 return {};
84 }
85 if (std::abs(to - from) <= parameterTolerance) {
86 return {SegmentIntersectionKind::Point, addScaled(a, r, (from + to) / 2.0), {}};
87 }
88 return {SegmentIntersectionKind::Overlap, addScaled(a, r, from), addScaled(a, r, to)};
89 }
90
91 const double t = crossVectors(cMinusA, s) / rCrossS;
92 const double u = crossVectors(cMinusA, r) / rCrossS;
93 const double tTolerance = tolerance / rLength;
94 const double uTolerance = tolerance / sLength;
95 if (t < -tTolerance || t > 1.0 + tTolerance ||
96 u < -uTolerance || u > 1.0 + uTolerance) {
97 return {};
98 }
99 return {SegmentIntersectionKind::Point, addScaled(a, r, std::clamp(t, 0.0, 1.0)), {}};
100}
