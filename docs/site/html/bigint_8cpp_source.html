23bool BigInteger::tryParse(const std::string& text, BigInteger* result)
24{
25 if (text.empty() || result == nullptr) return false;
26 std::size_t begin = 0;
27 int sign = 1;
28 if (text[begin] == '+' || text[begin] == '-') {
29 sign = text[begin++] == '-' ? -1 : 1;
30 }
31 if (begin == text.size()) return false;
32 for (std::size_t i = begin; i < text.size(); ++i) {
33 if (text[i] < '0' || text[i] > '9') return false;
34 }
35 BigInteger value;
36 for (std::size_t end = text.size(); end > begin;) {
37 const std::size_t start = end >= begin + 9 ? end - 9 : begin;
38 value.m_digits.push_back(static_cast<std::uint32_t>(std::stoul(text.substr(start, end - start))));
39 end = start;
40 }
41 value.m_sign = sign;
42 value.normalize();
43 *result = std::move(value);
44 return true;
45}

85BigInteger& BigInteger::operator*=(const BigInteger& other)
86{
87 if (m_sign == 0 || other.m_sign == 0) { m_digits.clear(); m_sign = 0; return *this; }
88 std::vector<std::uint64_t> product(m_digits.size() + other.m_digits.size());
89 for (std::size_t i = 0; i < m_digits.size(); ++i) {
90 std::uint64_t carry = 0;
91 for (std::size_t j = 0; j < other.m_digits.size(); ++j) {
92 const std::uint64_t value = product[i + j] +
93 static_cast<std::uint64_t>(m_digits[i]) * other.m_digits[j] + carry;
94 product[i + j] = value % kBase;
95 carry = value / kBase;
96 }
97 product[i + other.m_digits.size()] += carry;
98 }
99 m_digits.assign(product.begin(), product.end());
100 std::uint64_t carry = 0;
101 for (std::size_t i = 0; i < m_digits.size(); ++i) {
102 const std::uint64_t value = m_digits[i] + carry;
103 m_digits[i] = static_cast<std::uint32_t>(value % kBase);
104 carry = value / kBase;
105 }
106 if (carry) m_digits.push_back(static_cast<std::uint32_t>(carry));
107 m_sign *= other.m_sign;
108 normalize();
109 return *this;
110}

152{
153 std::int64_t borrow = 0;
154 for (std::size_t i = 0; i < m_digits.size(); ++i) {
155 std::int64_t value = static_cast<std::int64_t>(m_digits[i]) -
156 (i < other.m_digits.size() ? other.m_digits[i] : 0) - borrow;
