26bool onSegment(const Point& a, const Point& b, const Point& p, double eps)
27{
28 const double tolerance = std::abs(eps);
29 // Вырожденный отрезок — одна точка. В этом случае проверяем
30 // расстояние до неё.
31 const double dx = b.x - a.x;
32 const double dy = b.y - a.y;
33 const double lengthSquared = dx * dx + dy * dy;
34 if (lengthSquared <= tolerance * tolerance) {
35 const double px = p.x - a.x;
36 const double py = p.y - a.y;
37 return px * px + py * py <= tolerance * tolerance;
38 }
39
40 // Проецируем P на AB и ограничиваем проекцию концами отрезка.
41 // В отличие от допуска к cross, это расстояние имеет те же единицы,
42 // что и координаты, поэтому удобно для экранного радиуса точки.
43 const double projection = ((p.x - a.x) * dx + (p.y - a.y) * dy) / lengthSquared;
44 const double t = std::clamp(projection, 0.0, 1.0);
45 const double closestX = a.x + t * dx;
46 const double closestY = a.y + t * dy;
47 const double distanceX = p.x - closestX;
48 const double distanceY = p.y - closestY;
49 return distanceX * distanceX + distanceY * distanceY <= tolerance * tolerance;
50}
