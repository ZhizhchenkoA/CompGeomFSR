301void GeometryPage::computeTurn()
302{
303 const geometry::Point& a = m_points[0];
304 const geometry::Point& b = m_points[1];
305 const geometry::Point& p = m_points[2];
306
307 const double value = geometry::cross(a, b, p);
308 const double abX = b.x - a.x;
309 const double abY = b.y - a.y;
310 const bool degenerateSegment = abX * abX + abY * abY <= 1e-18;
311 // cross имеет размерность «пиксель²». Преобразуем радиус маркера в
312 // соответствующий допуск: расстояние до AB * длина AB.
313 const int sign = geometry::turn(a, b, p,
314 kVisualPointTolerance * std::hypot(abX, abY));
315
316 QString text;
317 QColor color;
318 if (degenerateSegment) {
319 text = "0 — A и B совпадают: направление не определено";
320 color = QColor(148, 163, 184); // серый
321 } else if (sign > 0) {
322 text = "1 — P слева от AB";
323 color = QColor(22, 163, 74); // зелёный
324 } else if (sign < 0) {
325 text = "−1 — P справа от AB";
326 color = QColor(220, 38, 38); // красный
327 } else {
328 // Коллинеарны: уточняем, на отрезке точка или на прямой вне его.
329 if (geometry::onSegment(a, b, p, kVisualPointTolerance)) {
330 text = "0 — P на отрезке AB";
331 color = QColor(234, 179, 8); // жёлтый
332 } else {
333 text = "0 — P на прямой (вне отрезка)";
334 color = QColor(148, 163, 184); // серый
335 }
336 }
337
338 ui->canvas->setPointColor(2, color);
339 ui->resultLabel->setText("Результат: " + text);
340 emit logMessage(QString("Предикат поворота: %1 (векторное произведение = %2)")
341 .arg(text)
342 .arg(value, 0, 'f', 2));
343}

403void GeometryPage::computeIntersection()
404{
405 if (m_points.size() != 4) {
406 return;
407 }
408
409 const geometry::SegmentIntersection result = geometry::intersectSegments(
410 m_points[0], m_points[1], m_points[2], m_points[3], kVisualPointTolerance);
411 ui->canvas->clearIntersectionVisuals();
412
413 switch (result.kind) {
414 case geometry::SegmentIntersectionKind::None:
415 ui->resultLabel->setText("Результат: пересечения нет");
416 emit logMessage("Пересечение отрезков: нет.");
417 break;
418 case geometry::SegmentIntersectionKind::Point: {
419 const QPointF point = geometry_qt::toScreen(result.first);
420 ui->canvas->setIntersectionMarker(point, true);
421 ui->resultLabel->setText(QString("Результат: (%1; %2)")
422 .arg(result.first.x, 0, 'f', 1)
423 .arg(result.first.y, 0, 'f', 1));
424 emit logMessage("Пересечение отрезков: точка.");
425 break;
426 }
427 case geometry::SegmentIntersectionKind::Overlap:
428 ui->canvas->setOverlapHighlight(geometry_qt::toScreen(result.first),
429 geometry_qt::toScreen(result.second), true);
430 ui->resultLabel->setText("Результат: общий участок");
431 emit logMessage("Пересечение отрезков: общий участок.");
432 break;
433 }

502void GeometryPage::computeDelaunay()
503{
504 const std::vector<geometry::DelaunayTriangle> triangulation =
505 geometry::delaunayTriangulation(m_points);
506 std::vector<std::array<QPointF, 3>> triangles;
507 std::vector<std::pair<QPointF, double>> circles;
508 triangles.reserve(triangulation.size());
509 circles.reserve(triangulation.size());
510 for (const geometry::DelaunayTriangle& triangle : triangulation) {
511 triangles.push_back({geometry_qt::toScreen(triangle.vertices[0]),
512 geometry_qt::toScreen(triangle.vertices[1]),
513 geometry_qt::toScreen(triangle.vertices[2])});
514 circles.emplace_back(geometry_qt::toScreen(triangle.circumcircle.center),
515 triangle.circumcircle.radius);
516 }
517 ui->canvas->setDelaunayTriangles(triangles, circles);
518 ui->resultLabel->setText(QString("Треугольников: %1").arg(triangles.size()));
519 emit logMessage(QString("Триангуляция Делоне: %1 треугольников.").arg(triangles.size()));
520}

558void GeometryPage::computePolygonOperation()
559{
560 const std::vector<geometry::Point> first = geometry::convexHull(m_polygonA);
561 const std::vector<geometry::Point> second = geometry::convexHull(m_polygonB);
562 geometry::ConvexPolygonOperation operation = geometry::ConvexPolygonOperation::Intersection;
563 if (m_currentTask == 8) {
564 operation = geometry::ConvexPolygonOperation::Union;
565 } else if (m_currentTask == 9) {
566 operation = geometry::ConvexPolygonOperation::Difference;
567 }
568 const geometry::PolygonSet result = geometry::convexPolygonBoolean(first, second, operation);
569
570 const auto toScreen = [](const std::vector<geometry::Point>& polygon) {
571 std::vector<QPointF> converted;
572 converted.reserve(polygon.size());
573 for (const geometry::Point& point : polygon) converted.push_back(geometry_qt::toScreen(point));
574 return converted;
575 };
576 std::vector<std::vector<QPointF>> screenResult;
577 screenResult.reserve(result.size());
578 for (const auto& contour : result) screenResult.push_back(toScreen(contour));
579 QColor resultColor(220, 38, 38); // пересечение
580 if (m_currentTask == 8) {
581 resultColor = QColor(124, 58, 237); // объединение
582 } else if (m_currentTask == 9) {
583 resultColor = QColor(225, 29, 72); // разность
584 }
585 ui->canvas->setPolygonOperation(toScreen(first), toScreen(second), screenResult, resultColor);
586 ui->resultLabel->setText(QString("Контуров результата: %1").arg(result.size()));
587 emit logMessage(QString("Операция над выпуклыми многоугольниками: %1 контуров.").arg(result.size()));
588}
