59 std::array<double, 6> values{};
60 for (int i = 0; i < 6; ++i) {

65 QString doubleResult = "не удалось преобразовать";
66 if (doubleOk) {

69 const int doubleSign = determinant > 0 ? 1 : determinant < 0 ? -1 : 0;
70 doubleResult = signText(doubleSign);
